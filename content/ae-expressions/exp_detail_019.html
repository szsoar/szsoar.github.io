---------------  方法一：  --------------- 
if (time >= 6) {
    key(numKeys).value;
} else {
    loopOut();
}

---------------  方法二：  --------------- 
const n = effect("Slider Control")("Slider");
const dur = key(numKeys).time - key(1).time;
const end = key(1).time + dur * n;

if (time >= end) {
    key(numKeys).value;
} else {
    loopOut();
}
