const colors = Array.from({ length: 5 }, () => [Math.random(), Math.random(), Math.random()]);
let txt = text.sourceText.style;
const count = text.sourceText.length;
Array.from({ length: count }, (_, i) => {
    txt = txt.setFillColor(colors[i % 5], i, 1);
});
txt;
